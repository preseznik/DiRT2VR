cmake_minimum_required(VERSION 3.24)
project(DiRT2VRLanLab LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "LAN lab requires MSVC Win32")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
get_filename_component(REPO "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_subdirectory("${REPO}/.deps/xlivelessness" xlln)
target_sources(XLiveLessNess PRIVATE "${CMAKE_CURRENT_LIST_DIR}/profile.cpp" "${CMAKE_CURRENT_LIST_DIR}/integration.cpp" "${CMAKE_CURRENT_LIST_DIR}/intro.cpp" "${REPO}/src/common.cpp")
target_sources(XLiveLessNess PRIVATE "${CMAKE_CURRENT_LIST_DIR}/browser.cpp")
target_include_directories(XLiveLessNess PRIVATE "${REPO}/.deps/xlivelessness/xlivelessness")
# Only D3DX is absent from the current Windows SDK. Do not shadow modern DXGI/D2D headers.
file(GLOB LEGACY_D3DX "${DXSDK_DIR}/Include/d3dx9*.h" "${DXSDK_DIR}/Include/d3dx9*.inl")
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/legacy-include")
file(COPY ${LEGACY_D3DX} DESTINATION "${CMAKE_BINARY_DIR}/legacy-include")
target_include_directories(XLiveLessNess PRIVATE "${REPO}/src" "${CMAKE_BINARY_DIR}/legacy-include")
target_compile_options(XLiveLessNess PRIVATE /FIstdlib.h)
set_property(TARGET XLiveLessNess PROPERTY CXX_STANDARD 20)
target_link_libraries(XLiveLessNess PRIVATE bcrypt)
target_link_options(XLiveLessNess PRIVATE /NODEFAULTLIB:MSVCRT)
add_executable(lan_profile_test "${REPO}/tests/lan_profile_test.cpp" profile.cpp)
target_include_directories(lan_profile_test PRIVATE "${CMAKE_CURRENT_LIST_DIR}")
target_compile_definitions(lan_profile_test PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_features(lan_profile_test PRIVATE cxx_std_20)
target_link_libraries(lan_profile_test PRIVATE shell32)
enable_testing()
add_test(NAME lan_profile_redirect COMMAND lan_profile_test)
add_executable(lan_intro_test "${REPO}/tests/lan_intro_test.cpp" intro.cpp)
target_include_directories(lan_intro_test PRIVATE "${CMAKE_CURRENT_LIST_DIR}")
target_compile_definitions(lan_intro_test PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_features(lan_intro_test PRIVATE cxx_std_20)
add_test(NAME lan_intro_guards COMMAND lan_intro_test)
add_executable(lan_browser_test "${REPO}/tests/lan_browser_test.cpp" browser.cpp)
target_include_directories(lan_browser_test PRIVATE "${REPO}/.deps/xlivelessness/xlivelessness")
target_compile_definitions(lan_browser_test PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN DIRT2VR_DISCOVERY_PORT=39821)
target_compile_features(lan_browser_test PRIVATE cxx_std_20)
target_link_libraries(lan_browser_test PRIVATE ws2_32)
add_test(NAME lan_browser_idle COMMAND lan_browser_test)
add_executable(lan_keepalive_test "${REPO}/tests/lan_keepalive_test.cpp")
target_compile_definitions(lan_keepalive_test PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_features(lan_keepalive_test PRIVATE cxx_std_20)
target_link_libraries(lan_keepalive_test PRIVATE ws2_32)
add_dependencies(lan_keepalive_test XLiveLessNess)
add_test(NAME lan_keepalive COMMAND lan_keepalive_test "$<TARGET_FILE:XLiveLessNess>" "${CMAKE_BINARY_DIR}/keepalive-fixture/xlln.ini")
set_tests_properties(lan_keepalive PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE)
