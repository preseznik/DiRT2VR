cmake_minimum_required(VERSION 3.24)
project(DiRT2VRLanLab LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "LAN lab requires MSVC Win32")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
get_filename_component(REPO "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_subdirectory("${REPO}/.deps/xlivelessness" xlln)
target_sources(XLiveLessNess PRIVATE "${CMAKE_CURRENT_LIST_DIR}/profile.cpp" "${CMAKE_CURRENT_LIST_DIR}/integration.cpp" "${REPO}/src/common.cpp")
# Only D3DX is absent from the current Windows SDK. Do not shadow modern DXGI/D2D headers.
file(GLOB LEGACY_D3DX "${DXSDK_DIR}/Include/d3dx9*.h" "${DXSDK_DIR}/Include/d3dx9*.inl")
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/legacy-include")
file(COPY ${LEGACY_D3DX} DESTINATION "${CMAKE_BINARY_DIR}/legacy-include")
target_include_directories(XLiveLessNess PRIVATE "${REPO}/src" "${CMAKE_BINARY_DIR}/legacy-include")
target_compile_options(XLiveLessNess PRIVATE /FIstdlib.h)
set_property(TARGET XLiveLessNess PROPERTY CXX_STANDARD 20)
target_link_libraries(XLiveLessNess PRIVATE bcrypt)
target_link_options(XLiveLessNess PRIVATE /NODEFAULTLIB:MSVCRT)
add_executable(lan_profile_test "${REPO}/tests/lan_profile_test.cpp" profile.cpp)
target_include_directories(lan_profile_test PRIVATE "${CMAKE_CURRENT_LIST_DIR}")
target_compile_definitions(lan_profile_test PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_features(lan_profile_test PRIVATE cxx_std_20)
target_link_libraries(lan_profile_test PRIVATE shell32)
enable_testing()
add_test(NAME lan_profile_redirect COMMAND lan_profile_test)
