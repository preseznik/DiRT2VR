cmake_minimum_required(VERSION 3.24)
project(DiRT2VR LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "DiRT2VR requires MSVC Win32 (configure with -A Win32)")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_subdirectory(.deps/minhook EXCLUDE_FROM_ALL)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_API_LAYERS OFF CACHE BOOL "" FORCE)
set(BUILD_CONFORMANCE_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_LOADER ON CACHE BOOL "" FORCE)
set(DYNAMIC_LOADER OFF CACHE BOOL "" FORCE)
add_subdirectory(.deps/OpenXR-SDK EXCLUDE_FROM_ALL)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/generate_exports.py"
  "$ENV{WINDIR}/SysWOW64/d3d11.dll" "${CMAKE_BINARY_DIR}" COMMAND_ERROR_IS_FATAL ANY)
add_library(d3d11 SHARED src/proxy.cpp src/trace.cpp src/common.cpp src/eye_pair.cpp src/eye_blit.cpp src/camera_math.cpp src/game_xr.cpp src/xr_frames.cpp src/vr_hotkeys.cpp "${CMAKE_BINARY_DIR}/exports.def")
target_include_directories(d3d11 PRIVATE "${CMAKE_BINARY_DIR}" src)
target_link_libraries(d3d11 PRIVATE minhook d3dcompiler dxguid psapi bcrypt openxr_loader)
target_compile_definitions(d3d11 PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX XR_USE_PLATFORM_WIN32 XR_USE_GRAPHICS_API_D3D11)
target_compile_options(d3d11 PRIVATE /W4 /EHsc)
add_executable(xr_probe src/xr_probe.cpp src/xr_frames.cpp src/xr_render_probe.cpp)
target_include_directories(xr_probe PRIVATE .deps/OpenXR-SDK/src/common)
target_link_libraries(xr_probe PRIVATE openxr_loader d3d11.lib dxgi d3dcompiler)
target_compile_definitions(xr_probe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX XR_USE_PLATFORM_WIN32 XR_USE_GRAPHICS_API_D3D11)
enable_testing()
add_test(NAME export_generator COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/test_exports.py")
add_executable(xr_frames_test tests/xr_frames_test.cpp src/xr_frames.cpp)
target_include_directories(xr_frames_test PRIVATE src .deps/OpenXR-SDK/include)
target_compile_definitions(xr_frames_test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX XR_USE_PLATFORM_WIN32 XR_USE_GRAPHICS_API_D3D11)
add_test(NAME xr_frame_lifecycle COMMAND xr_frames_test)
add_executable(eye_pair_test tests/eye_pair_test.cpp src/eye_pair.cpp src/eye_blit.cpp)
target_include_directories(eye_pair_test PRIVATE src)
target_compile_definitions(eye_pair_test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(eye_pair_test PRIVATE d3d11.lib d3dcompiler)
add_test(NAME eye_pair_gpu COMMAND eye_pair_test)
add_executable(camera_math_test tests/camera_math_test.cpp src/camera_math.cpp)
target_include_directories(camera_math_test PRIVATE src .deps/OpenXR-SDK/include)
add_test(NAME camera_math COMMAND camera_math_test)
add_executable(vr_hotkeys_test tests/vr_hotkeys_test.cpp src/vr_hotkeys.cpp)
target_include_directories(vr_hotkeys_test PRIVATE src)
target_compile_definitions(vr_hotkeys_test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
add_test(NAME vr_hotkeys COMMAND vr_hotkeys_test)
